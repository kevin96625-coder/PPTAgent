'use strict';
const fs = require('node:fs');
const { spawnSync } = require('node:child_process');

const CHROME_CANDIDATES = [
  '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome',
  '/Applications/Chromium.app/Contents/MacOS/Chromium',
  '/usr/bin/google-chrome',
  '/usr/bin/chromium',
  '/usr/bin/chromium-browser',
];

function findChrome() {
  for (const p of CHROME_CANDIDATES) {
    try { if (fs.existsSync(p)) return p; } catch { /* ignore */ }
  }
  const which = spawnSync('bash', ['-lc', 'command -v google-chrome || command -v chromium || command -v chromium-browser'], { encoding: 'utf8' });
  const out = (which.stdout || '').trim();
  return out ? out.split('\n')[0] : null;
}

function buildRenderArgs(svgPath, pngPath) {
  return [
    '--headless', '--disable-gpu', '--no-sandbox', '--hide-scrollbars',
    `--screenshot=${pngPath}`, '--window-size=1280,720',
    `file://${svgPath}`,
  ];
}

// Returns {ok:boolean, reason?:string}. Never throws.
function renderSvg(svgPath, pngPath, chrome = findChrome()) {
  if (!chrome) return { ok: false, reason: 'chrome-not-found' };
  const r = spawnSync(chrome, buildRenderArgs(svgPath, pngPath), { encoding: 'utf8' });
  if (r.status !== 0) return { ok: false, reason: (r.stderr || 'render-failed').slice(0, 200) };
  if (!fs.existsSync(pngPath)) return { ok: false, reason: 'no-output' };
  return { ok: true };
}

module.exports = { findChrome, buildRenderArgs, renderSvg, CHROME_CANDIDATES };
```

'use strict';
const IMAGE_STATUSES = new Set(['fetched', 'generated', 'missing']);

// kind: brand-entity|scene ; status derived from path + kind unless explicitly set.
function makeEntry({ page, kind, path = null, source_url = null, backend = null, width = null, height = null, status = null }) {
  let s = status;
  if (!s) {
    if (!path) s = 'missing';
    else s = kind === 'scene' ? 'generated' : 'fetched';
  }
  return { page, kind, path, source_url, backend, width, height, status: s };
}

function validateManifest(m) {
  const errors = [];
  if (!m || typeof m !== 'object' || !Array.isArray(m.images)) {
    return { ok: false, errors: ['manifest.images must be an array'] };
  }
  m.images.forEach((e, i) => {
    if (typeof e.page !== 'number') errors.push(`images[${i}].page must be number`);
    if (!IMAGE_STATUSES.has(e.status)) errors.push(`images[${i}].status invalid: ${e.status}`);
    if (e.status !== 'missing' && !e.path) errors.push(`images[${i}] non-missing requires path`);
  });
  return { ok: errors.length === 0, errors };
}

module.exports = { makeEntry, validateManifest, IMAGE_STATUSES };
```

'use strict';
// Flatten nested parts[].pages[] (same shape the validator reads).
function collectPages(outline) {
  const pages = [];
  for (const part of (outline.parts || [])) {
    if (!part || typeof part !== 'object') continue;
    for (const pg of (part.pages || [])) if (pg && typeof pg === 'object') pages.push(pg);
  }
  return pages;
}
// Cross-reference outline asset needs against the images manifest.
function assetPlan(outline, manifest = { images: [] }) {
  const pages = collectPages(outline);
  const byPage = new Map((manifest.images || []).map((e) => [e.page, e]));
  const asset_pages = pages
    .filter((pg) => pg && pg.asset && pg.asset.needed === true)
    .map((pg) => ({ page: pg.index, kind: pg.asset.kind, query: pg.asset.query || null }));
  const needs_real_image = asset_pages
    .filter(({ page }) => {
      const e = byPage.get(page);
      return !e || e.status === 'missing' || !e.path;
    })
    .map(({ page }) => page);
  return { asset_pages, needs_real_image };
}
module.exports = { assetPlan };
```

#!/usr/bin/env bash
# Best-effort image search+download. NEVER blocks caller.
# usage: image-search.sh "<query>" "<out_path>"
set -uo pipefail
QUERY="${1:-}"; OUT="${2:-}"
if [[ -z "$QUERY" || -z "$OUT" ]]; then echo "usage: image-search.sh <query> <out_path>" >&2; exit 2; fi

# 1) Try registered MCP image search via mcporter (e.g. exa), if available.
if command -v mcporter >/dev/null 2>&1; then
  SERVERS="$(mcporter list 2>/dev/null || true)"
  if echo "$SERVERS" | grep -qi exa; then
    URL="$(mcporter call 'exa.image_search' --query "$QUERY" 2>/dev/null | grep -oE 'https?://[^" ]+\.(jpg|jpeg|png|webp)' | head -1 || true)"
    if [[ -n "${URL:-}" ]] && curl -fsSL "$URL" -o "$OUT" 2>/dev/null; then echo "$OUT"; exit 0; fi
  fi
fi

# 2) Fallback: Bing image search results page scrape via curl (best-effort, may be empty).
ENC="$(printf '%s' "$QUERY" | sed 's/ /+/g')"
HTML="$(curl -fsSL -A 'Mozilla/5.0' "https://www.bing.com/images/search?q=${ENC}" 2>/dev/null || true)"
URL="$(printf '%s' "$HTML" | grep -oE 'murl&quot;:&quot;[^&]+' | head -1 | sed 's/murl&quot;:&quot;//')"
if [[ -n "${URL:-}" ]] && curl -fsSL -A 'Mozilla/5.0' "$URL" -o "$OUT" 2>/dev/null; then
  # minimal sanity: file exists and >10KB (skip thumbnails/blanks)
  if [[ -f "$OUT" && $(wc -c <"$OUT") -gt 10240 ]]; then echo "$OUT"; exit 0; fi
fi

echo "no-image-source" >&2
exit 1
```
