'use strict';
function hexToRgb(hex) {
  let h = String(hex).trim().replace(/^#/, '');
  if (h.length === 3) h = h.split('').map((c) => c + c).join('');
  if (!/^[0-9a-fA-F]{6}$/.test(h)) return null;
  return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)];
}
function rgbToHex([r, g, b]) {
  const c = (n) => Math.max(0, Math.min(255, Math.round(n))).toString(16).padStart(2, '0');
  return `#${c(r)}${c(g)}${c(b)}`;
}
function relLuminance(hex) {
  const rgb = hexToRgb(hex);
  if (!rgb) return 0;
  const [r, g, b] = rgb.map((v) => {
    const s = v / 255;
    return s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4);
  });
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}
function contrastRatio(a, b) {
  const la = relLuminance(a), lb = relLuminance(b);
  const hi = Math.max(la, lb), lo = Math.min(la, lb);
  return (hi + 0.05) / (lo + 0.05);
}
// Move `fill` toward white (lighten) or black (darken) in small steps until contrast vs bg >= target.
function adjustToContrast(fill, bg, target, dir) {
  if (contrastRatio(fill, bg) >= target) return fill;
  const rgb = hexToRgb(fill);
  if (!rgb) return fill;
  let cur = rgb.slice();
  for (let i = 0; i < 40; i++) {
    cur = cur.map((v) => dir === 'lighten' ? v + (255 - v) * 0.1 : v - v * 0.1);
    const hex = rgbToHex(cur);
    if (contrastRatio(hex, bg) >= target) return hex;
  }
  return dir === 'lighten' ? '#ffffff' : '#000000';
}
module.exports = { hexToRgb, rgbToHex, relLuminance, contrastRatio, adjustToContrast };
```

'use strict';
const { contrastRatio, adjustToContrast, relLuminance } = require('./color-contrast');

// Operate on <text>/<tspan> tags via regex (no XML dep; project convention).
function autofixSvg(svg, opts = {}) {
  const fontFloor = opts.fontFloor || 14;
  const labelFloor = opts.labelFloor || 12;
  const fixes = [];
  let out = svg;

  // --- font floor: raise any font-size below the applicable floor ---
  out = out.replace(/<(text|tspan)\b[^>]*>/g, (tag) => {
    const m = tag.match(/font-size="([0-9.]+)"/);
    if (!m) return tag;
    const size = parseFloat(m[1]);
    const isLabel = /data-role="(label|footnote)"/.test(tag);
    const floor = isLabel ? labelFloor : fontFloor;
    if (size >= floor) return tag;
    fixes.push({ type: 'font-floor', detail: `${size}->${floor}` });
    return tag.replace(/font-size="[0-9.]+"/, `font-size="${floor}"`);
  });

  return { svg: out, fixes };
}
module.exports = { autofixSvg };
```

```javascript
  // --- safe-area: only top-level (no transform/g ancestor) role-tagged title/page-number ---
  function topLevelTextTags(s) {
    // collect index ranges covered by any <g ...> ... </g> (nesting-aware)
    const ranges = [];
    const stack = [];
    const tokRe = /<g\b[^>]*>|<\/g>/g; let t;
    while ((t = tokRe.exec(s))) {
      if (t[0] === '</g>') { const o = stack.pop(); if (o != null) ranges.push([o, t.index]); }
      else stack.push(t.index);
    }
    const inG = (i) => ranges.some(([a, b]) => i > a && i < b);
    return { inG };
  }
  const { inG } = topLevelTextTags(out);
  out = out.replace(/<text\b[^>]*>/g, (tag, idx) => {
    if (inG(idx)) return tag; // nested -> never touch coords
    const role = (tag.match(/data-role="([^"]+)"/) || [])[1];
    if (role !== 'title' && role !== 'page-number') return tag;
    let t2 = tag;
    const ym = tag.match(/\by="([0-9.]+)"/);
    const xm = tag.match(/\bx="([0-9.]+)"/);
    if (role === 'title' && ym && parseFloat(ym[1]) < 92) {
      t2 = t2.replace(/\by="[0-9.]+"/, 'y="92"'); fixes.push({ type: 'safe-area', detail: 'title y->92' });
    }
    if (role === 'page-number') {
      if (ym && parseFloat(ym[1]) > 688) { t2 = t2.replace(/\by="[0-9.]+"/, 'y="688"'); fixes.push({ type: 'safe-area', detail: 'pagenum y->688' }); }
      if (xm && parseFloat(xm[1]) > 1220) { t2 = t2.replace(/\bx="[0-9.]+"/, 'x="1220"'); fixes.push({ type: 'safe-area', detail: 'pagenum x->1220' }); }
    }
    return t2;
  });
```
